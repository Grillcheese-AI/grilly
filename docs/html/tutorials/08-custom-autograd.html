Custom ReLU

import numpy as np
 from grilly.nn.autograd import Function, FunctionCtx, Variable
 
 class MyReLU(Function):
     @staticmethod
     def forward(ctx: FunctionCtx, x: Variable) -> np.ndarray:
         ctx.save_for_backward(x)
         return np.maximum(x.data, 0)
 
     @staticmethod
     def backward(ctx: FunctionCtx, grad_output: np.ndarray):
         x, = ctx.saved_tensors
         return grad_output * (x.data > 0).astype(np.float32)

Custom Linear

class MyLinear(Function):
     @staticmethod
     def forward(ctx, x, weight, bias):
         ctx.save_for_backward(x, weight, bias)
         return x.data @ weight.data.T + bias.data
 
     @staticmethod
     def backward(ctx, grad_output):
         x, weight, bias = ctx.saved_tensors
         grad_x = grad_output @ weight.data
         grad_w = grad_output.T @ x.data
         grad_b = grad_output.sum(axis=0)
         return grad_x, grad_w, grad_b

Training with Custom Functions

from grilly.nn.autograd import tensor, randn, no_grad
 
 x = tensor([[-1, 0, 1]], requires_grad=False)
 w = randn(2, 3, requires_grad=True)
 b = randn(2, requires_grad=True)
 
 y = MyLinear.apply(x, w, b)
 loss = y.sum()
 loss.backward()
 
 with no_grad():
     w.data -= 0.01 * w.grad
     b.data -= 0.01 * b.grad
